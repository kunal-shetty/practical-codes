1)create a package with a function and procedure to find the sum of first 10 natural numbers.

create or replace package natural_nos
as
procedure sumno(a in int);
function sumon (a int) return int;
end natural_nos;
/

create or replace package body natural_nos
as
procedure sumno (a in int)
as 
begin 
dbms_output.put_line((a*(a+1))/2);
end;

function sumon (a int) return int
as
begin
return (a*(a+1))/2;
end;
end natural_nos;
/

execute natural_nos.sumno(10);

select natural_nos.sumon(10) from dual;

 FUNCTION TOTEMP RETURN INT;
END;

CREATE OR REPLACE PACKAGE BODY PACK_DEMO
AS
 PROCEDURE TOTEMP
 AS 
  T INT;
 BEGIN
  SELECT COUNT(*) INTO T FROM EMP WHERE JOB='CLERK';

 END;
 FUNCTION TOTEMP RETURN INT

  T INT;
